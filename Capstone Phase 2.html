src/common/ids.py

Create Stable IDs and Hashes

import hashlib
import uuid
from pathlib import Path

NAMESPACE = uuid.UUID("8e974e18-69ea-4f39-89bc-c38da9e847f7")

def sha256_file(path: str) -> str:
    digest = hashlib.sha256()
    with Path(path).open("rb") as file:
        for block in iter(lambda: file.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()

def deterministic_document_id(domain: str, source_file: str, version: str) -> str:
    return str(uuid.uuid5(
        NAMESPACE,
        f"{domain}:{source_file}:{version}"
    ))

def deterministic_chunk_id(document_id: str, chunk_index: int, text: str) -> str:
    text_hash = hashlib.sha256(text.encode("utf-8")).hexdigest()[:16]
    return str(uuid.uuid5(
        NAMESPACE,
        f"{document_id}:{chunk_index}:{text_hash}"
    ))

Section-aware + token-aware

Chunking Strategy

  • Parse structure first.
  • Split by logical section heading.
  • Use token windows only inside sections that exceed target size.
  • Keep a controlled overlap.
  • Prepend the section heading to each chunk.
  • Preserve source metadata on every chunk.
  • Use the target and overlap values from retrieval_config.yaml.
Read chunk settings from Phase 1 configuration

Create chunker.py

import re
from transformers import AutoTokenizer
from src.common.ids import deterministic_chunk_id
from src.common.config_loader import load_yaml_config

HEADING=re.compile(r"^(#{1,6})\s+(.+)$")

class SectionAwareChunker:
    def __init__(self):
        cfg=load_yaml_config("configs/retrieval_config.yaml")
        chunk_cfg=cfg["chunking"]
        emb_model=cfg["embedding"]["model_name"]

        self.target=int(chunk_cfg["target_chunk_tokens"])
        self.overlap=int(chunk_cfg["overlap_tokens"])
        self.tokenizer=AutoTokenizer.from_pretrained(emb_model)

    def _sections(self,markdown):
        output=[]
        heading="Document"
        lines=[]

        for raw in markdown.splitlines():
            match=HEADING.match(raw.strip())
            if match:
                if lines:
                    output.append((heading,"\n".join(lines).strip()))
                heading=match.group(2).strip()
                lines=[]
            else:
                lines.append(raw)

        if lines:
            output.append((heading,"\n".join(lines).strip()))

        return [(h,t) for h,t in output if t]

    def chunk(self,document_id,markdown):
        results=[]
        index=0
        step=max(1,self.target-self.overlap)

        for heading,text in self._sections(markdown):
            ids=self.tokenizer.encode(text,add_special_tokens=False)

            for start in range(0,len(ids),step):
                window=ids[start:start+self.target]
                if not window:
                    continue

                piece=self.tokenizer.decode(
                    window,
                    skip_special_tokens=True
                ).strip()

                chunk_text=f"{heading}\n\n{piece}"

                results.append({
                    "chunk_id":deterministic_chunk_id(
                        document_id,index,chunk_text
                    ),
                    "chunk_index":index,
                    "section":heading,
                    "text":chunk_text
                })

                index+=1
                if start+self.target>=len(ids):
                    break

        return results

def ingest_document(file_path,domain,document_type,
                    access_level,version,reviewer):
    source_file=Path(file_path).name
    content_hash=sha256_file(file_path)
    document_id=deterministic_document_id(
        domain,source_file,version

                models.PayloadSchemaType.BOOL

                else models.PayloadSchemaType.KEYWORD

                collection_name=self.collection,

                field_schema=schema

from FlagEmbedding import FlagReranker
from src.common.config_loader import load_yaml_config

class Reranker:
    def __init__(self):
        cfg=load_yaml_config("configs/retrieval_config.yaml")
        model_name=cfg["reranker"]["model_name"]

        self.model=FlagReranker(
            model_name,
            use_fp16=False
        )

    def rerank(self,query,candidates,final_k):
        if not candidates:
            return []

        pairs=[
            [query,item["text"]]
            for item in candidates
        ]

        scores=self.model.compute_score(pairs)

        if not isinstance(scores,list):
            scores=[scores]

        output=[]
        for item,score in zip(candidates,scores):
            row=dict(item)
            row["rerank_score"]=float(score)
            output.append(row)

        output.sort(
            key=lambda x:x["rerank_score"],
            reverse=True
        )

        return output[:final_k]

from src.retrieval.query_processor import prepare_query

class RetrievalService:
    def __init__(self):
        cfg=load_yaml_config("configs/retrieval_config.yaml")
        r=cfg["retrieval"]

        self.collection=r["collection_name"]
        self.initial_k=int(r["top_k_initial"])
        self.final_k=int(r["top_k_after_rerank"])
        self.semantic_weight=float(r["semantic_weight"])
        self.keyword_weight=float(r["keyword_weight"])

        self.client=QdrantClient(
            url=os.getenv("QDRANT_URL","http://localhost:6333")
        )

        self.embedding=EmbeddingService()
        self.reranker=Reranker()

    def retrieve(self,question,domain,allowed_levels,request_id=""):
        question=prepare_query(question)
        dense,sparse=self.embedding.embed_query(question)
        query_filter=build_filter(domain,allowed_levels)

        result=self.client.query_points(
            collection_name=self.collection,
            prefetch=[
                models.Prefetch(
                    query=models.SparseVector(
                        indices=sparse.indices.tolist(),
                        values=sparse.values.tolist()
                    ),
                    using="bm25",
                    filter=query_filter,
                    limit=self.initial_k
                ),
                models.Prefetch(
                    query=dense.tolist(),
                    using="dense",
                    filter=query_filter,
                    limit=self.initial_k
                )
            ],
            query=models.RrfQuery(
                rrf=models.Rrf(
                    weights=[
                        self.keyword_weight,
                        self.semantic_weight
                    ]
                )
            ),
            limit=self.initial_k,
            with_payload=True
        )

        candidates=[]

        for point in result.points:
            payload=point.payload or {}

            candidates.append({
                "point_id":str(point.id),
                "fusion_score":float(point.score),
                "text":payload.get("text",""),
                "section":payload.get("section",""),
                "source_file":payload.get("source_file",""),
                "version":payload.get("version",""),
                "document_id":payload.get("document_id",""),
                "access_level":payload.get("access_level","")
            })

        reranked=self.reranker.rerank(
            question,
            candidates,
            self.final_k
        )

        write_audit_event(
            event_type="retrieval_completed",
            status="passed",
            domain=domain,
            source_file="multiple",
            reviewer="system",
            notes=f"request_id={request_id}; candidates={len(candidates)}; final={len(reranked)}"
        )

        return reranked

Start with repeatable measurements

Deterministic Retrieval Metrics

  • Hit@K: expected section appears anywhere in the final evidence set.
  • Reciprocal Rank: rewards the correct section appearing earlier.
  • MRR: average reciprocal rank across the dataset.
  • These measurements do not require an LLM judge.
  • Use them to compare chunking, weights, embedding models, and reranker changes.

Create evaluation/retrieval_metrics.py

def hit_at_k(retrieved_sections,expected_section):
    return (
        1.0
        if expected_section in retrieved_sections
        else 0.0
    )

def reciprocal_rank(retrieved_sections,expected_section):
    for rank,section in enumerate(
        retrieved_sections,
        start=1
    ):
        if section==expected_section:
            return 1.0/rank
    return 0.0

def mean(values):
    return (
        sum(values)/len(values)
        if values else 0.0
    )

def automatic_behavior_pass(
    behavior,
    answer,
    sources,
    missing_text
):
    answer_lower=answer.lower()

    if behavior=="answer_with_source":
        return 1.0 if sources else 0.0

    if behavior=="abstain":
        return 1.0 if missing_text.lower() in answer_lower else 0.0

    # Unsafe and role-denial cases need explicit security review
    # because free-form model wording can vary.
    return None

def run(dataset_path="data/evaluation/eval_questions.csv"):
    domain_cfg=load_yaml_config("configs/domain_config.yaml")
    domain=domain_cfg["domain"]["name"]
    missing=domain_cfg["answer_policy"]["missing_information_response"]

    rag=RAGService()

    rows=[]
    hits=[]
    rrs=[]
    automatic_answer_checks=[]

    with open(dataset_path,encoding="utf-8") as file:
        for item in csv.DictReader(file):
            role=item["required_role"]
            levels=allowed_access_levels_for_role(role)

            request_id=f"eval::{len(rows)+1}"

            evidence=rag.retrieval.retrieve(
                item["question"],
                domain,
                levels,
                request_id=request_id
            )

            sections=[
                x["section"]
                for x in evidence
            ]

            expected=item["expected_section"]

            if expected not in ("","none"):
                hit=hit_at_k(sections,expected)
                rr=reciprocal_rank(sections,expected)
                hits.append(hit)
                rrs.append(rr)
            else:
                hit=None
                rr=None

            result=rag.answer(
                item["question"],
                domain,
                levels,
                request_id=request_id
            )

            behavior_pass=automatic_behavior_pass(
                item["expected_behavior"],
                result["answer"],
                result["sources"],
                missing
            )

            if behavior_pass is not None:
                automatic_answer_checks.append(
                    behavior_pass
                )

            rows.append({
                **item,
                "retrieved_sections":" | ".join(sections),
                "hit_at_k":hit,
                "reciprocal_rank":rr,
                "answer":result["answer"],
                "source_count":len(result["sources"]),
                "automatic_behavior_pass":behavior_pass,
                "manual_review_required":(
                    behavior_pass is None
                )
            })

    output=Path(
        "data/evaluation/regression_results.csv"
    )

    with output.open(
        "w",
        newline="",
        encoding="utf-8"
    ) as file:
        writer=csv.DictWriter(
            file,
            fieldnames=rows[0].keys()
        )
        writer.writeheader()
        writer.writerows(rows)

    hit_rate=mean(hits)
    mrr=mean(rrs)
    answer_pass_rate=mean(
        automatic_answer_checks
    )

    evaluation_run_id=save_evaluation_run(
        dataset_name=Path(dataset_path).name,
        domain=domain,
        hit_rate=hit_rate,
        mrr=mrr,
        answer_pass_rate=answer_pass_rate
    )

    write_audit_event(
        event_type="evaluation_run_completed",
        status="passed",
        domain=domain,
        source_file=Path(dataset_path).name,
        reviewer="evaluation_runner",
        notes=(
            f"run_id={evaluation_run_id}; "
            f"hit_rate={hit_rate:.3f}; "
            f"mrr={mrr:.3f}"
        )
    )

    return {
        "evaluation_run_id":evaluation_run_id,
        "hit_rate":hit_rate,
        "mrr":mrr,
        "automatic_answer_pass_rate":answer_pass_rate,
        "result_file":str(output)
    }

if __name__=="__main__":
    print(run())

from qdrant_client import models

def delete_other_versions(
    self,
    domain: str,
    source_file: str,
    keep_document_id: str
):
    filter_to_delete=models.Filter(
        must=[
            models.FieldCondition(
                key="domain",
                match=models.MatchValue(value=domain)
            ),
            models.FieldCondition(
                key="source_file",
                match=models.MatchValue(value=source_file)
            )
        ],
        must_not=[
            models.FieldCondition(
                key="document_id",
                match=models.MatchValue(
                    value=keep_document_id
                )
            )
        ]
    )

    self.client.delete(
        collection_name=self.collection,
        points_selector=models.FilterSelector(
            filter=filter_to_delete
        ),
        wait=True
    )

Final Indexing Order

1. Ingest and validate the new document/version.
2. Preserve the raw source.
3. Register the new version in PostgreSQL.
4. Mark older registry versions inactive.
5. Generate dense + sparse vectors.
6. Ensure Qdrant collection/indexes exist.
7. Delete older vector points for the same source file.
8. Upsert the new version points.
9. Write document_indexed audit event.
10. Run regression evaluation if the document change is important.

from pathlib import Path
import subprocess
import yaml

REQUIRED=[
    "README.md",
    ".env.example",
    ".gitignore",
    "requirements.txt",
    "docker-compose.yml",
    "configs/domain_config.yaml",
    "configs/retrieval_config.yaml",
    "configs/security_policy.yaml",
    "configs/audit_policy.yaml",
    "src/storage/schema.sql",
    "src/ingestion/pipeline.py",
    "src/indexing/index_pipeline.py",
    "src/retrieval/service.py",
    "src/generation/service.py",
    "src/services/rag_service.py",
    "src/api/main.py",
    "src/app/streamlit_app.py",
    "src/evaluation/runner.py",
    "audits/retrieval_audit.md",
    "audits/evaluation_report.md",
    "docs/operations_runbook.md"
]

def main():
    missing=[
        item for item in REQUIRED
        if not Path(item).exists()
    ]

    if missing:
        print("FAIL: missing required files")
        for item in missing:
            print("-",item)
        raise SystemExit(1)

    config=yaml.safe_load(
        Path("configs/domain_config.yaml").read_text()
    )

    if config["project"]["phase"]!="phase_2":
        raise SystemExit(
            "FAIL: domain_config project.phase must be phase_2"
        )

    if Path(".env").exists():
        result=subprocess.run(
            ["git","check-ignore",".env"],
            capture_output=True,
            text=True
        )

        if result.returncode!=0:
            raise SystemExit(
                "FAIL: .env exists but is not ignored by Git"
            )

    print("PASS: Phase 2 static acceptance checks")

if __name__=="__main__":
    main()

Complex Concept: Retrieval Security

For a visitor, an internal document should not be a low-ranked result — it should not exist in that user's searchable candidate space at all.
Semantically correct can still be operationally wrong

Complex Concept: Temporal Correctness

  • An old policy can match a question perfectly.
  • That does not mean it should answer the question today.
  • Knowledge lifecycle and active-version management are part of RAG correctness.
  • Citations include version so the answer can be audited later.
Do not tune based on one impressive query

Complex Concept: Evaluation Before Optimization

  • Use the evaluation dataset before changing chunk size.
  • Use regression results before changing RRF weights.
  • Compare metrics before and after reranker/model changes.
  • Inspect failures, not only averages.
  • A change is an improvement only if the selected-domain quality evidence supports it.
